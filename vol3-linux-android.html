55.1 sysroot 是什么

# 编译器默认在 /usr/lib 找头文件和库 —— 那是主机的世界！
# sysroot = 「假装自己是目标板的根目录」：

55.2 pkg-config 交叉陷阱

PKG_CONFIG_PATH=/opt/sysroot-imx6/usr/lib/pkgconfig \
PKG_CONFIG_LIBDIR=/opt/sysroot-imx6/usr/lib/pkgconfig:/opt/sysroot-imx6/usr/share/pkgconfig \
pkg-config --cflags --libs openssl
# 关键： PKG_CONFIG_LIBDIR 必须设置以屏蔽主机路径泄漏！
# 否则 cmake 会把主机的 .pc 结果混进来 → 「本机能跑板上段错误」的经典根源
CMake 正确姿势： toolchain file 里 set(ENV{PKG_CONFIG_LIBDIR} ...)

55.3 动态链接细节四连

主题要点
版本符号libfoo.so.1.2.3 → SONAME libfoo.so.1；运行时找的是 SONAME 不是软链名
rpath vs LD_LIBRARY_PATH产品用 $ORIGIN 相对 rpath（-Wl,-rpath,'$ORIGIN/../lib'），别依赖环境变量
静态链接-static 全静态：部署零依赖但体积大+glibc NSS 功能残缺；musl 下体验更好
strip 与调试分离objcopy --only-keep-debug 出 .debug 文件，板上留 strip 后小文件，崩溃时主机配对解析(ch64)

55.6 原理深挖：动态链接器的搜索顺序真相

ld-linux 解析 NEEDED 的查找序列：
① DT_RPATH(旧,继承) → ② LD_LIBRARY_PATH 环境变量
→ ③ DT_RUNPATH(新,不继承) → ④ /etc/ld.so.cache → ⑤ 默认目录(/lib,/usr/lib)
嵌入式要点：
· 只信 $ORIGIN 相对 rpath —— 目标机环境变量不可控
· ldconfig 生成的 cache 在只读系统上不可用 → 直接静态链接或精确 rpath
· musl 的 loader 行为略有差异且无 ldconfig —— 全静态是 musl 甜点

55.7 完整工程：一条命令验证依赖闭环

#!/bin/sh # checkdeps.sh —— 上板前的体检
APP=$1; SYSROOT=$2
file $APP | grep -q ARM || echo "架构不对!"
readelf -d $APP | grep NEEDED | awk '{print $5}' | tr -d '[]' | while read lib; do
  found=$(find $SYSROOT -name "$lib*" | head -1)
  [ -z "$found" ] && echo "缺失: $lib"
done
# CI 里跑它，缺库在打包阶段就死而不是现场救火

55.8 进阶话题三则

  • -fdebug-prefix-map 的团队价值：把本机绝对路径映射为统一路径——core dump 与 perf 符号在不同机器间通用；

56.2 initramfs 三种用途

  1. 过渡跳板：真 rootfs 在 LVM/加密分区/NFS 上时，先在内存里装好驱动再 switch_root；
  2. 恢复环境：独立 recovery 系统（ch72 Android A/B 的 recovery 同思想）；
  3. 极速启动：整个系统就是 initramfs（内存盘），适合只读小型设备——开机即就绪无 IO 等待。
# 手工构建最小 initramfs：
mkdir -p initramfs/{bin,dev,proc,sys}
cp busybox initramfs/bin/ ; ln -s ../bin/busybox initramfs/bin/sh
cat > initramfs/init <<'EOF'
#!/bin/sh
mount -t proc none /proc; mount -t sysfs none /sys; mount -t devtmpfs none /dev
exec switch_root /newroot /sbin/init      # 挂真root后移交
EOF
find initramfs | cpio -o -H newc | gzip > initramfs.cpio.gz

static irqreturn_t my_hardirq(int irq,void*d){

    disable_hw_interrupt();            // 配合ONESHOT语义
    return IRQ_WAKE_THREAD;
}
static irqreturn_t my_threaded(int irq,void*d){
    process_data(priv);                // 可能耗几ms
    enable_hw_interrupt(priv);
    return IRQ_HANDLED;
}

int ep = epoll_create1(EPOLL_CLOEXEC);

for (;;) {
    int n = epoll_wait(ep, evs, MAX_EV, timeout_ms);
    for (i = 0; i < n; i++) {
        if (evs[i].data.fd == listenfd) {         /* ET下必须accept到EAGAIN！ */
            while ((c = accept4(listenfd,NULL,NULL,SOCK_NONBLOCK)) >= 0) add_ep(ep,c);

            /* ET 读也必须循环到 EAGAIN； LT 模式则可按需读 */
            while ((r = read(fds[i],buf,sizeof buf)) > 0) handle(buf,r);
            if (r == 0) close_and_del();           /* 对端关闭 */
        }
    }
}

for(i..n){ ioctl(fd,VIDIOC_QUERYBUF,&buf); bufs[i]=mmap(...); }
/* ③ 入队启动 */
for(i..n) ioctl(fd,VIDIOC_QBUF,...);

ioctl(fd,VIDIOC_QBUF,&buf);              /* 还回去继续填 */

68.5 原理深挖：Zygote fork 的三个约束（为什么 App 崩溃模式如此独特）

约束原因对开发的影响
只允单线程 fork复制多线程状态不安全zygote 预加载期禁止起线程——你的库若在静态块起线程会在 fork 后死锁
COW 共享页省内存核心机制写热点对象会触发页复制——大数组初始化放 App 自己做而非依赖共享
GC 状态一致性fork 时堆快照冻结zygote 预加载完成后必须停 GC 再 fork(源码有专门处理)

70.5 原理深挖：一次 oneway 调用的时序与风险面

oneway(异步单向)：
  客户端立即返回 → 事务进入目标 node 的 async 队列 → 目标线程池空闲取出
  风险①： async 队列有全局配额(~一半缓冲) —— 洪水期返回 ENOMEM
  风险②： 不同目标间无顺序保证；同一目标内保序 —— 「先发后至」错觉多源于跨对象
  风险③： 服务端处理慢会反压客户端吗？不会！—— 所以要有自己的流控
同步调用：
  占用客户端 binder 线程直至回包 —— 嵌套跨进程调用成环即死锁(ch48 锁序的进程版)

class HalClient {
  sp<IBinder> binder; sp<IMySensor> svc;
public:
  status_t connect(){
    auto sm = defaultServiceManager();
    binder = sm->getService(String16("myco.mysensor"));
    if(!binder) return TIMED_OUT;
    binder->linkToDeath(new DeathCb(*this), nullptr);   // 死亡自动重连钩子
    svc = interface_cast<IMySensor>(binder);

    for(int i=0;i<3;i++){
      if(svc) { int v; if(svc->readRaw(ch,&v).isOk()) return v; }
      usleep(50'000); connect();                         // 退避重建
    } return INT_MIN;
  }};

71.5 原理深挖：SELinux 类型强制(TC)的判定流水线

72.1 三代升级架构对照

架构机制痛点
Recovery 传统式进 recovery 小系统解包写 system 分区升级期间设备不可用；断电易变砖
A/B 无缝双 system slot 后台下载写入，重启切换分区空间翻倍

72.2 Virtual A/B 数据面

核心角色：
update_engine        # 用户态引擎： 下载/校验/写目标slot
dm-snapshot/dm-user  # 内核态： 写时重定向到 COW 空间
metadata (LP)        # 动态分区元数据记录 slot 组与 snapshot 状态
boot_control HAL     # bootloader 询问/标记 active/successful slot

升级流：
① 后台下载 delta 包(bsdiff/puffin 二进制差量) → 校验哈希
② 写 target slot： 未占用块引用源，改动块入 COW
③ merge 阶段(重启后 idle 时)： snapshot 合并回真分区
④ bootctl markBootSuccessful —— 若 N 次未确认则 bootloader 自动切回旧 slot
⑤ rollback 安全网： merge 前掉电=快照仍在可恢复；merge 中掉电=状态机续作

③ 工程结论

选型而非站队
poll / select 适合
连接少、每轮几乎都有事件、实现简单；select 还有 1024 fd 上限，poll 没有但仍需每次拷入内核并线性扫描。
epoll 适合
大量长连接、事件稀疏的服务器；关注列表常驻内核，返回的是 ready 集合，用户态只处理活跃连接。
别忘了 LT / ET
LT 未处理完会持续通知，编程简单；ET 只在状态变化时通知一次，必须配合非阻塞 fd 与“读到 EAGAIN”的循环，否则会丢事件。

高频追问 · 先答后看

6 问
用户态调用 read() 后，数据是怎样从硬件到达缓冲区的？哪里可能睡眠？

syscall 进入 VFS，按文件类型调用文件系统或字符设备的 read，驱动再从硬件 FIFO/DMA 取数据并 copy_to_user。若数据未就绪，驱动在等待队列上睡眠；copy_to_user 遇到未驻留的用户页也可能缺页睡眠。睡眠只允许发生在进程上下文。

为什么 copy_to_user 可能睡眠？哪些上下文不能调用它？

用户缓冲区可能尚未映射到物理页，访问会触发缺页处理，需要调度与 IO，因此可能睡眠。中断、软中断、持自旋锁等原子上下文不能睡眠，也就不能直接调用 copy_to_user；应先把数据放到内核缓冲，回到进程上下文再拷贝。

threaded_irq、workqueue、tasklet 的上下文与适用场景有什么不同？

threaded_irq 在内核线程上下文运行，可以睡眠，适合 I²C/SPI 等慢设备；workqueue 也是进程上下文，可延迟、可合并、可指定 CPU；tasklet 在软中断上下文运行，不可睡眠，只适合短小的原子收尾。实时系统常用 threaded irq 并设置线程优先级。

epoll 的水平触发与边缘触发分别怎么正确使用？

LT 只要 fd 可读就会持续通知，编程简单但可能重复处理；ET 只在状态变化时通知一次，必须用非阻塞 fd 循环读到 EAGAIN，否则会丢事件。ET 还要处理 EPOLLHUP/EPOLLERR，多线程场景可用 EPOLLEXCLUSIVE 或单 reactor 降低惊群。

Binder 一次调用从客户端到服务端，中间发生了什么？

客户端通过代理对象发起 transact，经 ioctl 进入 /dev/binder；内核驱动把事务写入目标进程的缓冲区并唤醒服务线程；服务线程执行 onTransact 后把结果写回。Binder 利用 mmap 减少一次拷贝，但线程池耗尽或服务阻塞都会表现为调用超时。

A/B OTA 在什么条件下回滚？回滚如何保证旧系统可用？

新槽启动失败计数达到阈值、签名/verity 校验失败、看门狗未被标记为启动成功，bootloader 就切回旧槽。旧槽在升级期间不被写入，元数据记录当前活动槽与成功标记；回滚后再上报失败原因，避免无限重试。
