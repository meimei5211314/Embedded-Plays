q15_t q15_mul(q15_t a, q15_t b){
    q31_t p = (q31_t)a * b;  /* Q30 */
    p >>= 15;
    if (p > 32767)  p = 32767;
    if (p < -32768) p = -32768;
    return (q15_t)p;
}

          定点滤波代码

q15_t lp_filter1(q15_t xn, q15_t *state,
                 q15_t alpha){
    q31_t d = ((q31_t)(xn - *state)) * alpha;
    *state = (q15_t)(*state + (d >> 15));
    return *state;
}

        
xn - state 是 Q15，乘 α 后变成 Q30；右移 15 位回到 Q15。α 越小，滤波越强，但跟随越慢；α 越大，跟随越快，但噪声也更容易通过。

        

          α raw 6554
          时间常数 ≈ 5 样本
        

        
若输入可能超出 Q15 范围，先在入口饱和；若状态可能积累，乘法中间量用 Q31。别把“溢出回绕”当成滤波特性。

      

    


    

      

        从滤波到 PID：定点控制的三条护栏
        扩展 · 第 53 章 FOC 会用到
      

      

        
误差与输出都用 Q15设定值与反馈值相减得到误差；P/I/D 三项先扩到 Q31 累加，最后统一饱和回 Q15，避免每步截断引入偏差。

        
积分项用 Q31，并抗饱和积分是“记忆”，位宽必须比比例项更宽；输出饱和时停止同方向积分，或用积分分离，防止退饱和时大幅超调。

        
微分作用在测量值上对误差求微分会在设定值阶跃时产生冲击；改为对反馈值求微分，并用低通滤波抑制噪声。

        
固定采样周期定点 PID 的积分/微分系数隐含采样周期；改变控制频率必须重新标定 Ki、Kd，不能直接复用旧参数。

      

      
/* 增量式 PID 的核心运算（示意，工程版需加完整抗饱和） */
q15_t pid_step(q15_t sp, q15_t fb){
    q15_t e = sp - fb;
    q31_t p = (q31_t)kp * e;          /* Q30 */
    integ += (q31_t)ki * e;           /* Q30 累加 */
    q15_t d = e - e_prev;
    q31_t out = (p >> 15) + (integ >> 15)
                 + (((q31_t)kd * d) >> 15);
    if (out > 32767)  { out = 32767;  integ -= (q31_t)ki * e; }
    if (out < -32768) { out = -32768; integ -= (q31_t)ki * e; }
    e_prev = e;
    return (q15_t)out;
}

3.7宏陷阱与故障排查：看见展开后的世界

      
宏的问题几乎都能用一句话解释：你看到的是一行，编译器看到的是一整段。遇到怪现象，先展开，再怀疑。

    

    

      

        
SQUARE(i++)：宏不是函数副作用 · UB

        

          
          
        

        

        

          

          

        

        
宏把参数替换两次，i++ 就执行两次；两个自增之间没有序列点，属于未定义行为。即使某次编译“算对了”，也不能依赖。

      

      

        
gcc -E：只看预处理排障第一板斧

        
$ gcc -E macro.c
# 等待运行：输出只经过预处理，不生成机器码

        

          
          
        

        
`-E` 能看到宏展开、条件编译和头文件展开，但看不到类型错误和运行时问题。它是“确认文本长什么样”，不是万能调试器。

        
宏调试三板斧：gcc -E 看展开 → #pragma message 注入版本/分支信息 → 临时改成同名 static inline 获得类型检查。

      

    

    

      
展开顺序：参数先展开，# / ## 后应用，只扫描一遍原理深挖

      

        
① 参数完全展开嵌套宏由内向外先展开；XSTR(__LINE__) 先得到 STR(42)。

        
② # / ## 才应用如果一开始就字符串化，拿到的是 "__LINE__"，不是行号。两级包装正是为了绕开这一点。

        
③ 结果只重扫一次本轮已展开的名字会被“画蓝”，不再递归。#define A B / #define B A 展开到 A 就停。

      

      
#define A B
#define B A

int x = A;   /* 展开一次后停回 A，而不是永远循环 */

      
“蓝色画布”是 GCC cpp 手册里的说法：本轮已经展开过的宏名会被标记，防止自引用死循环。看到奇怪的自引用结果，先想这条规则。

    

    

      
故障排障：四个高频现场点击展开

      

        

          宏展开后语法错乱 / else 悬空
          
根因：多语句宏没有 do-while 包裹，参数没有括号。定位：gcc -E 看展开结果；先补括号，再用 do { ... } while (0) 包住多语句。

        

        

          定点滤波输出跳变、溢出
          
根因：中间量位数不够、缺少饱和、系数超出 Q15 范围。定位：用 ±满量程、0、-1 做边界单测；打印中间 Q30 值；优先复用 CMSIS-DSP 的饱和函数。

        

        

          同一份代码，两块板子行为不同
          
根因：端序不同、结构体对齐不同、编译器版本/优化等级不同。定位：打印 sizeof、offsetof 和端序自检；统一工具链版本；协议边界用 memcpy + 明确字节序。

        

        

          X-Macro 新增项后链接错误
          
根因：枚举数量、字符串表长度、查表边界不同步；或 def 文件漏了续行反斜杠。定位：检查 \；用 _Static_assert(sizeof(err_str)/sizeof(char*) == ERR_COUNT, "desync") 在编译期兜底。

        

      

    

  


  
  

    

      
3.8平台差异、实测与实战 Lab

      
同样是 Q15 乘法，在 M0、M4F、RISC-V 上的代价并不一样。最后用一组编译期断言和边界测试，把本章变成可复现的证据。

    

    

      

        
Q15 vs 浮点：F407 上的数量级原课程实测 · -O2

        

          

            
操作（1000 次均值）Q15 定点float（FPU）double（软浮点）

            
              
一次乘加 MAC~6 cycles~4 cycles（VMLA）~120 cycles（库调用）

              
一阶 IIR 单步~18 cycles~14 cycles>800 cycles

              
1024 点 FFT（CMSIS）q15 ~11 µsf32 ~14 µs—

            
          

        

        
结论不是“定点永远更快”。在 M4F 上 float 已经很快；选定点的核心理由是确定性 + 可移植到 M0/无 FPU 型号 + 无异常路径。高频信号链可以定点，UI/统计用浮点。

      

      

        
平台差异：同一份 C，不同代价选型感知

        

          
Cortex-M0无 FPU、无硬件除法、无桶形移位器；变量移位和 double 都很贵。定点是默认选项，热循环里少做变长移位。

          
Cortex-M4F单精度 FPU + DSP 指令；__SSAT/__USAT 一条完成饱和，smlabb 可做 16×16+32 饱和累加。float 不慢，但定点更确定。

          
RV32IM有 M 扩展才有硬件乘除；无 M 时乘法走软库，Q15 的 16×16 乘法可能比模拟 32×32 更划算。先看芯片手册，再谈优化。

        

        
优化顺序永远是：先证明热点（profiler）→ 再换算法/数据类型 → 最后才手写内联汇编或 CMSIS-DSP。不要凭感觉“定点一定快”。
